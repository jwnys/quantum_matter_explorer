# Optional separate environment for the vLLM inference server, which is launched
# as a subprocess by src/llm/vllm.py.
#
# The simple setup is a single environment: pip install -e ".[vllm]" installs the
# same pins via pyproject.toml and the launcher then uses the current interpreter.
# Use this file only if you want the server isolated in its own environment
# (useful on clusters, where the two stacks are easier to rebuild separately):
#   python3 -m venv <path-to-server-venv>
#   <path-to-server-venv>/bin/pip install -U pip
#   <path-to-server-venv>/bin/pip install -r requirements-server.txt
# then point config/config.yaml llm.vllm.server_python at
# <path-to-server-venv>/bin/python (or export QME_VLLM_PYTHON).
#
# Note for cluster installs: torch wheels are large, so if /tmp is small set TMPDIR
# and PIP_CACHE_DIR to a roomy disk before installing.

# vLLM pins its own torch / transformers / xformers transitively, so only vLLM is
# pinned here and it resolves the rest for the target Python. (Pinning torch
# alongside can force an incompatible build.)
vllm==0.21.0
# The FlashInfer attention kernels vLLM uses on recent NVIDIA GPUs. Pinned to the
# version this release was tested with.
flashinfer-python==0.6.8.post1
